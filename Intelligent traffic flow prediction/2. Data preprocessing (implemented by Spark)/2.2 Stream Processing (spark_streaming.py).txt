from pyspark.sql import SparkSession
from pyspark.sql.types import *
from pyspark.sql.functions import from_json

spark = SparkSession.builder.appName("TrafficStreaming").getOrCreate()

# Define the schema for the stream data
schema = StructType([
    StructField("sensor_id", StringType()),
    StructField("speed", DoubleType()),
    StructField("volume", IntegerType()),
    StructField("timestamp", TimestampType())
])

# Read stream data from Kafka
stream_df = spark.readStream \
    .format("kafka") \
    .option("kafka.bootstrap.servers", "kafka:9092") \
    .option("subscribe", "traffic_updates") \
    .load() \
    .selectExpr("CAST(value AS STRING) as json") \
    .select(from_json("json", schema).alias("data")) \
    .select("data.*")

# Real-time processing
def process_batch(batch_df, batch_id):
    batch_df.write.mode("append").parquet("hdfs://traffic_data/stream_processed/")

stream_df.writeStream \
    .foreachBatch(process_batch) \
    .start() \
    .awaitTermination()
