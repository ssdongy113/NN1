import torch
import torch.nn as nn

class TrafficLSTM(nn.Module):
    def __init__(self, input_size=6, hidden_size=64, output_size=3):
        """
        Initialize the TrafficLSTM model.

        Parameters:
        - input_size: Number of input features.
        - hidden_size: Size of the hidden state.
        - output_size: Number of output features.
        """
        super(TrafficLSTM, self).__init__()
        self.lstm = nn.LSTM(input_size, hidden_size, batch_first=True)  # LSTM layer
        self.fc = nn.Linear(hidden_size, output_size)  # Fully connected layer

    def forward(self, x):
        """
        Forward pass through the TrafficLSTM model.

        Parameters:
        - x: Input data with shape (batch_size, seq_len, input_size).

        Returns:
        - Output tensor after passing through the LSTM and fully connected layer.
        """
        # x shape: (batch_size, seq_len, input_size)
        lstm_out, _ = self.lstm(x)  # lstm_out shape: (batch_size, seq_len, hidden_size)
        return self.fc(lstm_out[:, -1, :])  # Only take the last time step
