4.1 信号灯优化 (signal_optimizer.py)
import cvxpy as cp
import numpy as np

def optimize_traffic_signals(predicted_flows, current_queues, green_limits=(15, 60)):
    """
    predicted_flows: 各方向预测流量数组 [veh/h]
    current_queues: 当前各方向排队长度 [veh]
    """
    num_phases = len(predicted_flows)
    t_green = cp.Variable(num_phases, pos=True)
    
    # 目标函数：最小化总延误
    objective = cp.Minimize(
        cp.sum_squares(current_queues + 0.5*predicted_flows*t_green/3600 - t_green/2) + 
        0.1*cp.sum_squares(t_green[1:] - t_green[:-1])
    )
    
    # 约束条件
    constraints = [
        cp.sum(t_green) <= 120,  # 总周期不超过120秒
        t_green >= green_limits[0],
        t_green <= green_limits[1]
    ]
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
    return np.round(t_green.value, 0)

# 示例使用
pred_flows = np.array([800, 600, 400, 300])  # 东西南北方向预测流量
queues = np.array([12, 8, 5, 3])  # 当前排队车辆数
optimal_greens = optimize_traffic_signals(pred_flows, queues)
print("优化后的绿灯时间(秒):", optimal_greens)
