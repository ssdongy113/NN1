多源数据采集(data_collection)

import requests
import pandas as pd
from datetime import datetime

class TrafficDataCollector:
    @staticmethod
    def get_sensor_data(api_url):
        """从交通传感器API获取实时数据"""
        response = requests.get(api_url)
        return pd.DataFrame(response.json()['data'])
    
    @staticmethod
    def get_weather_data(lat, lon, api_key):
        """获取OpenWeatherMap数据"""
        url = f"https://api.openweathermap.org/data/2.5/weather?lat={lat}&lon={lon}&appid={api_key}"
        return requests.get(url).json()
    
    @staticmethod
    def get_historical_traffic(city, days):
        """模拟从数据库获取历史数据"""
        # 实际应用中替换为SQL查询或Hadoop Hive操作
        return pd.read_parquet(f"hdfs://{city}_traffic_history.parquet")

# 示例使用
collector = TrafficDataCollector()
realtime_data = collector.get_sensor_data("http://traffic-api.la.gov/sensors")
weather_data = collector.get_weather_data(34.05, -118.25, "your_api_key")
